function SetNoclip(state)
    NoclipEnabled = state
    if state then
        NoclipConn = RunService.Stepped:Connect(function()
            local char = player.Character
            if char and NoclipEnabled then
                for _, part in pairs(char:GetDescendants()) do
                    if part:IsA("BasePart") then part.CanCollide = false end
                end
            end
        end)
    else
        if NoclipConn then NoclipConn:Disconnect(); NoclipConn = nil end
        local char = player.Character
        if char then
            for _, part in pairs(char:GetDescendants()) do
                if part:IsA("BasePart") then part.CanCollide = true end
            end
        end
    end
end

player.CharacterAdded:Connect(function()
    if NoclipEnabled then task.wait(0.5); SetNoclip(true) end
end)

local function getWaterPart()
    if not WalkWaterPart or not WalkWaterPart.Parent then

end

local function isHoldingPortalFruit()
    local char = player.Character
    if not char then return false end
    local tool = char:FindFirstChildOfClass("Tool")
    if tool then
        local tName = string.lower(tool.Name)
        if string.find(tName, "portal") or string.find(tName, "door") then return true end
    end
    return false
end

local function equipSanguineArt()
    local char = player.Character
    if not char then return nil end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum then return nil end
    local tool = char:FindFirstChild("Sanguine Art") or char:FindFirstChild("Sanguine")
    if not tool then
        local bp = player:FindFirstChild("Backpack")
        if bp then
            for _, t in ipairs(bp:GetChildren()) do
                local name = string.lower(t.Name)
                if string.find(name, "sanguine") then
                    hum:EquipTool(t)
                    tool = t
                    task.wait(0.1)
                    break
                end
            end
        end
    end
    return tool
end

local function equipPortalFruit()
    local char = player.Character
    if not char then return nil end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum then return nil end
    local tool = char:FindFirstChild("Portal-Portal") or char:FindFirstChild("Portal Fruit") or char:FindFirstChild("Portal") or char:FindFirstChild("Door")
    if not tool then
        local bp = player:FindFirstChild("Backpack")
        if bp then
            for _, t in ipairs(bp:GetChildren()) do
                local name = string.lower(t.Name)
                if string.find(name, "portal") or string.find(name, "door") then
                    hum:EquipTool(t)
                    tool = t
                    task.wait(0.1)
                    break
                end
            end
        end
    end
    return tool
end

local function doPortalCombo()
    if not isHoldingPortalFruit() then
        equipPortalFruit()
        task.wait(0.08)
    end
    if not isHoldingPortalFruit() then return end
    local function pressKey(kc)
        VirtualInputManager:SendKeyEvent(true, kc, false, game)
        task.wait(0.05)
        VirtualInputManager:SendKeyEvent(false, kc, false, game)
    end
    pressKey(Enum.KeyCode.X)
    task.wait(0.15)
    pressKey(Enum.KeyCode.Z)
end

    VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.C, false, game)

end

-- ============================================================
local function startAntiLava()
    if antiLavaConnection then antiLavaConnection:Disconnect() end
    local antiLavaTimer = 0
    antiLavaConnection = RunService.Stepped:Connect(function(_, dt)
        antiLavaTimer = antiLavaTimer + dt
        if antiLavaTimer < 0.2 then return end
        antiLavaTimer = 0
        local char = player.Character
        if not (char and antiLavaActive) then return end
        for _, part in pairs(char:GetDescendants()) do
            if part:IsA("BasePart") and part.Name ~= "HumanoidRootPart"
            and part.Name ~= "Torso" and part.Name ~= "UpperTorso"
            and part.Name ~= "LowerTorso" and part.Name ~= "Head" then
                part.CanTouch = false
            end
        end
    end)
end

local function stopAntiLava()
    if antiLavaConnection then antiLavaConnection:Disconnect(); antiLavaConnection = nil end
end

-- SILENT AIM SIMPLE (HOOK FIRESERVER)
-- ============================================================
task.spawn(function()
    while true do
        task.wait(0.5)
        if not silentAimEnabled then cachedTarget = nil; continue end
        pcall(function()
            local hrp = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
            if not hrp then cachedTarget = nil; return end
            local closest, closestDist = nil, selectedRange
            for _, p in ipairs(Players:GetPlayers()) do
                if p ~= player and p.Character then
                    if not teamCheck or (player.Team ~= p.Team) then
                        local eHRP = p.Character:FindFirstChild("HumanoidRootPart")
                        local hum = p.Character:FindFirstChildOfClass("Humanoid")
                        if eHRP and hum and hum.Health > 0 then
                            local d = (eHRP.Position - hrp.Position).Magnitude
                            if d < closestDist then closest = p; closestDist = d end
                        end
                    end
                end
            end
            cachedTarget = closest
        end)
    end
end)

local function installSilentAim()
    if silentInstalled then return end
    silentInstalled = true
    task.spawn(function()
        while true do
            task.wait(0.1)
            if silentAimEnabled and cachedTarget then
                pcall(function()
                    local eHRP = cachedTarget.Character and cachedTarget.Character:FindFirstChild("HumanoidRootPart")
                    silentPos = eHRP and eHRP.Position or nil
                end)
            else
                silentPos = nil
            end
        end
    end)
    task.spawn(function()
        local mt = getrawmetatable(game)
        local oldNC = mt.__namecall
        setreadonly(mt, false)
        mt.__namecall = newcclosure(function(...)
            local method = getnamecallmethod()
            local args = {...}
            if silentAimEnabled and silentPos and tostring(method) == "FireServer" and tostring(args[1]) == "RemoteEvent" then
                if typeof(args[2]) == "Vector3" then args[2] = silentPos; return oldNC(unpack(args))
                elseif typeof(args[2]) == "CFrame" then args[2] = CFrame.new(silentPos); return oldNC(unpack(args)) end
            end
            return oldNC(...)
        end)
        setreadonly(mt, true)
    end)
end
installSilentAim()

                        bill.Size = UDim2.new(1, 200, 1, 30)
                        bill.Adornee = v.Character.Head
                        bill.AlwaysOnTop = true
                        local name = Instance.new('TextLabel', bill)
                        name.Font = Enum.Font.Code
                        name.FontSize = "Size14"
                        name.TextWrapped = true
                        name.Size = UDim2.new(1, 0, 1, 0)
                        name.TextYAlignment = 'Top'
                        name.BackgroundTransparency = 1
                        name.TextStrokeTransparency = 0.5
                        name.Text = (v.Name .. ' \n' .. round((player.Character.Head.Position - v.Character.Head.Position).Magnitude / 3) .. ' M')
                        if v.Team == player.Team then name.TextColor3 = Color3.new(0, 0, 254) else name.TextColor3 = Color3.new(255, 0, 0) end
                    else
                        if v.Character.Head:FindFirstChild('NameEsp'..Number) then

                            label.Text = (v.Name .. ' | ' .. round((player.Character.Head.Position - v.Character.Head.Position).Magnitude / 3) .. ' M\nHealth : ' .. round(v.Character.Humanoid.Health * 100 / v.Character.Humanoid.MaxHealth) .. '%')

                else
                    if v.Character.Head:FindFirstChild('NameEsp'..Number) then v.Character.Head:FindFirstChild('NameEsp'..Number):Destroy() end

            end
        end)
    end

local function createPage(isScrollable)
    local page = Instance.new("Frame")
    page.Name = "Page"
    page.BackgroundTransparency = 1
    page.Size = UDim2.fromScale(1, 1)
    page.Visible = false
    page.Parent = ContentBody

    local left, right
    if isScrollable then
        left = Instance.new("ScrollingFrame")
        right = Instance.new("ScrollingFrame")
        for _, scroller in ipairs({left, right}) do
            scroller.BackgroundTransparency = 1
            scroller.BorderSizePixel = 0
            scroller.ScrollBarThickness = 3
            scroller.ScrollBarImageColor3 = COLORS.Accent
            scroller.CanvasSize = UDim2.new()
            scroller.AutomaticCanvasSize = Enum.AutomaticSize.Y
            scroller.Parent = page
        end
    else
        left = Instance.new("Frame")
        right = Instance.new("Frame")
        left.BackgroundTransparency = 1
        right.BackgroundTransparency = 1
        left.Parent = page
        right.Parent = page
    end

    left.Position = UDim2.fromScale(0, 0)
    left.Size = UDim2.new(0.5, -5, 1, 0)
    right.Position = UDim2.new(0.5, 5, 0, 0)
    right.Size = UDim2.new(0.5, -5, 1, 0)

    for _, column in ipairs({left, right}) do
        local layout = Instance.new("UIListLayout")
        layout.Padding = UDim.new(0, 6)
        layout.SortOrder = Enum.SortOrder.LayoutOrder
        layout.Parent = column
    end

    return page, left, right
end
