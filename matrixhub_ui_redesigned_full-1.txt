function SetNoclip(state)
    NoclipEnabled = state
    if state then
        NoclipConn = RunService.Stepped:Connect(function()
            local char = player.Character
            if char and NoclipEnabled then
                for _, part in pairs(char:GetDescendants()) do
                    if part:IsA("BasePart") then part.CanCollide = false end
                end
            end
        end)
    else
        if NoclipConn then NoclipConn:Disconnect(); NoclipConn = nil end
        local char = player.Character
        if char then
            for _, part in pairs(char:GetDescendants()) do
                if part:IsA("BasePart") then part.CanCollide = true end
            end
        end
    end
end

player.CharacterAdded:Connect(function()
    if NoclipEnabled then task.wait(0.5); SetNoclip(true) end
end)

local function isHoldingPortalFruit()
    local char = player.Character
    if not char then return false end
    local tool = char:FindFirstChildOfClass("Tool")
    if tool then
        local tName = string.lower(tool.Name)
        if string.find(tName, "portal") or string.find(tName, "door") then return true end
    end
    return false
end

local function equipSanguineArt()
    local char = player.Character
    if not char then return nil end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum then return nil end
    local tool = char:FindFirstChild("Sanguine Art") or char:FindFirstChild("Sanguine")
    if not tool then
        local bp = player:FindFirstChild("Backpack")
        if bp then
            for _, t in ipairs(bp:GetChildren()) do
                local name = string.lower(t.Name)
                if string.find(name, "sanguine") then
                    hum:EquipTool(t)
                    tool = t
                    task.wait(0.1)
                    break
                end
            end
        end
    end
    return tool
end

local function equipPortalFruit()
    local char = player.Character
    if not char then return nil end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum then return nil end
    local tool = char:FindFirstChild("Portal-Portal") or char:FindFirstChild("Portal Fruit") or char:FindFirstChild("Portal") or char:FindFirstChild("Door")
    if not tool then
        local bp = player:FindFirstChild("Backpack")
        if bp then
            for _, t in ipairs(bp:GetChildren()) do
                local name = string.lower(t.Name)
                if string.find(name, "portal") or string.find(name, "door") then
                    hum:EquipTool(t)
                    tool = t
                    task.wait(0.1)
                    break
                end
            end
        end
    end
    return tool
end

local function doPortalCombo()
    if not isHoldingPortalFruit() then

    end

end

local function startAntiLava()
    if antiLavaConnection then antiLavaConnection:Disconnect() end
    local antiLavaTimer = 0
    antiLavaConnection = RunService.Stepped:Connect(function(_, dt)
        antiLavaTimer = antiLavaTimer + dt
        if antiLavaTimer < 0.2 then return end
        antiLavaTimer = 0
        local char = player.Character
        if not (char and antiLavaActive) then return end
        for _, part in pairs(char:GetDescendants()) do
            if part:IsA("BasePart") and part.Name ~= "HumanoidRootPart"
            and part.Name ~= "Torso" and part.Name ~= "UpperTorso"
            and part.Name ~= "LowerTorso" and part.Name ~= "Head" then
                part.CanTouch = false
            end
        end
    end)
end

local function stopAntiLava()
    if antiLavaConnection then antiLavaConnection:Disconnect(); antiLavaConnection = nil end
end

task.spawn(function()
    while true do
        task.wait(0.5)
        if not silentAimEnabled then cachedTarget = nil; continue end
        pcall(function()
            local hrp = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
            if not hrp then cachedTarget = nil; return end
            local closest, closestDist = nil, selectedRange
            for _, p in ipairs(Players:GetPlayers()) do
                if p ~= player and p.Character then
                    if not teamCheck or (player.Team ~= p.Team) then
                        local eHRP = p.Character:FindFirstChild("HumanoidRootPart")
                        local hum = p.Character:FindFirstChildOfClass("Humanoid")
                        if eHRP and hum and hum.Health > 0 then
                            local d = (eHRP.Position - hrp.Position).Magnitude
                            if d < closestDist then closest = p; closestDist = d end
                        end
                    end
                end
            end
            cachedTarget = closest
        end)
    end
end)

local function installSilentAim()
    if silentInstalled then return end
    silentInstalled = true
    task.spawn(function()
        while true do
            task.wait(0.1)
            if silentAimEnabled and cachedTarget then
                pcall(function()
                    local eHRP = cachedTarget.Character and cachedTarget.Character:FindFirstChild("HumanoidRootPart")
                    silentPos = eHRP and eHRP.Position or nil
                end)
            else
                silentPos = nil
            end
        end
    end)
    task.spawn(function()
        local mt = getrawmetatable(game)
        local oldNC = mt.__namecall
        setreadonly(mt, false)
        mt.__namecall = newcclosure(function(...)
            local method = getnamecallmethod()
            local args = {...}
            if silentAimEnabled and silentPos and tostring(method) == "FireServer" and tostring(args[1]) == "RemoteEvent" then
                if typeof(args[2]) == "Vector3" then args[2] = silentPos; return oldNC(unpack(args))
                elseif typeof(args[2]) == "CFrame" then args[2] = CFrame.new(silentPos); return oldNC(unpack(args)) end
            end
            return oldNC(...)
        end)
        setreadonly(mt, true)
    end)
end
installSilentAim()

local MacroEnabled = false

local macroGui = nil

local function macroPressKey(kc, holdTime)
    if not kc then return end
    VirtualInputManager:SendKeyEvent(true, kc, false, game)
    task.wait(holdTime or 0.05)
    VirtualInputManager:SendKeyEvent(false, kc, false, game)
end

local function macroSafeEquip(slotNum)
    local slotKeys = { [1] = Enum.KeyCode.One, [2] = Enum.KeyCode.Two, [3] = Enum.KeyCode.Three, [4] = Enum.KeyCode.Four }
    local kc = slotKeys[slotNum]
    if not kc then return end
    macroPressKey(kc)
    task.wait(0.05)
    local char = player.Character
    if not (char and char:FindFirstChildOfClass("Tool")) then
        macroPressKey(kc)
        task.wait(0.05)
    end
end

local function executeMacroCombo()
    local slots = {
        {MacroSlot1, MacroKey1},
        {MacroSlot2, MacroKey2},
        {MacroSlot3, MacroKey3},
        {MacroSlot4, MacroKey4},
    }
    for i, data in ipairs(slots) do
        local slot, key = data[1], data[2]
        if key and key ~= "OFF" then
            if i == 1 or slots[i-1][1] ~= slot then macroSafeEquip(slot) end
            local kc = Enum.KeyCode[key]
            if kc then macroPressKey(kc) end
            if i < #slots then task.wait(MacroDelay) end
        end
    end
end

local function isFlashstepMacro(track)
    local names={"FlashStepRegular","FlashStepDraco","FlashStep","Flashstep","Soru"}
    local ids={"17555632156","18461649274","616006778","616010882","5403485593","5403491911"}
    local n=string.lower(track.Name or "")
    for _,name in ipairs(names) do if string.find(n,string.lower(name)) then return true end end
    local id=track.Animation and tostring(track.Animation.AnimationId):match("%d+") or ""
    for _,wanted in ipairs(ids) do if id==wanted then return true end end
    return false
end

function executeSoruCombo()
    if not MacroEnabled or MacroMode ~= "Soru" or MacroExecuting then return end
    MacroExecuting=true
    pcall(function()
        macroSafeEquip(MacroSlot1); task.wait(0.06); if Enum.KeyCode[MacroKey1] then macroPressKey(Enum.KeyCode[MacroKey1],0.08) end
        task.wait(MacroDelay)
        if MacroSlot1~=MacroSlot2 then macroSafeEquip(MacroSlot2); task.wait(0.06) end; if Enum.KeyCode[MacroKey2] then macroPressKey(Enum.KeyCode[MacroKey2],0.08) end
        task.wait(MacroDelay)
        if MacroSlot2~=MacroSlot3 then macroSafeEquip(MacroSlot3); task.wait(0.06) end; if Enum.KeyCode[MacroKey3] then macroPressKey(Enum.KeyCode[MacroKey3],0.08) end
        task.wait(MacroDelay)
        if MacroSlot3~=MacroSlot4 then macroSafeEquip(MacroSlot4); task.wait(0.06) end; if Enum.KeyCode[MacroKey4] then macroPressKey(Enum.KeyCode[MacroKey4],0.08) end
    end)
    MacroExecuting=false
end

    local hum=char:WaitForChild("Humanoid",5); if not hum then return end
    hum.AnimationPlayed:Connect(function(track)
        if MacroEnabled and MacroMode=="Soru" and isFlashstepMacro(track) then task.spawn(executeSoruCombo) end
    end)
end
